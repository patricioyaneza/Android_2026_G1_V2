Ejemplo usando binding 

atributo de clase
	private val productos = mutableListOf("")
o también puede ser:
	val elementos = listOf("Elemento 1", "Elemento 2", "Elemento 3", "Elemento 4")


EN EVENTO onCreate


	binding.btnAceptar.setOnClickListener {

            var producto = binding.txtNombre.text.toString().trim().uppercase()
            if (producto.isEmpty()) {
                return@setOnClickListener
            }
            binding.txtNombre.setText("")
            this.productos.add(producto)
            val adapter = ArrayAdapter(this, R.layout.simple_list_item_1, productos)
            binding.listadoInformacion.adapter = adapter
        }

        // Manejo de eventos de clic
        binding.listadoInformacion.setOnItemClickListener { _, _, position, _ ->
            val seleccionado = productos[position]
            Toast.makeText(this, "Seleccionaste: $seleccionado", Toast.LENGTH_SHORT).show()
        }
