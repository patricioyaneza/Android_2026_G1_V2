3.- en Activity java:
	(fab solo, es como un botón)

    private FloatingActionButton fabPrincipal;
    private FloatingActionButton fabCalculadora;
    private FloatingActionButton fabConfiguracion;
    private FloatingActionButton fabAdicional;

// permite almacenar el estado de apertura visibilidad de los botones
private boolean abierto = false;

// dentro de onCreate 

        // Referencias a los botones
        fabPrincipal = findViewById(R.id.fabPrincipal);
        fabCalculadora = findViewById(R.id.fabCalculadora);
        fabConfiguracion = findViewById(R.id.fabConfiguracion);
        fabAdicional = findViewById(R.id.fabAdicional);

        // Mostrar u ocultar las opciones
        fabPrincipal.setOnClickListener(v -> {

            if (abierto) {
                fabCalculadora.setVisibility(View.GONE);
                fabConfiguracion.setVisibility(View.GONE);
                fabAdicional.setVisibility(View.GONE);
            } else {
                fabCalculadora.setVisibility(View.VISIBLE);
                fabConfiguracion.setVisibility(View.VISIBLE);
                fabAdicional.setVisibility(View.VISIBLE);
            }

            abierto = !abierto;
        });

        // Acción del botón Calculadora
        fabCalculadora.setOnClickListener(v -> {

            Toast.makeText(this, "Calculadora", Toast.LENGTH_SHORT).show();
        });

        // Acción del botón Configuración
        fabConfiguracion.setOnClickListener(v -> {
            Toast.makeText(this, "Configuración", Toast.LENGTH_SHORT).show();
        });
